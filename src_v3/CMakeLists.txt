cmake_minimum_required(VERSION 3.18)
project(Robot_MPC_v3 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
list(APPEND CMAKE_PREFIX_PATH "/opt/openrobots")
find_package(Eigen3 REQUIRED)
find_package(pinocchio REQUIRED)
find_package(qpOASES REQUIRED)
find_package(Threads REQUIRED)

add_library(mpc_core_v3 STATIC
    ../src_v2/pinocchio_fun.cpp dynamics_backend.cpp prepared_model.cpp
    prepared_qp.cpp mpc_controller_v3.cpp)
set_target_properties(mpc_core_v3 PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(mpc_core_v3 PUBLIC
    "${CMAKE_CURRENT_SOURCE_DIR}/../include_v3" "${CMAKE_CURRENT_SOURCE_DIR}/../include_v2"
    ${qpOASES_INCLUDE_DIR} ${qpOASES_INCLUDE_DIRS})
target_link_libraries(mpc_core_v3 PUBLIC Eigen3::Eigen pinocchio::pinocchio Threads::Threads)
if(TARGET qpOASES::qpOASES)
    target_link_libraries(mpc_core_v3 PUBLIC qpOASES::qpOASES)
elseif(TARGET qpOASES)
    target_link_libraries(mpc_core_v3 PUBLIC qpOASES)
elseif(qpOASES_LIBRARIES)
    target_link_libraries(mpc_core_v3 PUBLIC ${qpOASES_LIBRARIES})
elseif(qpOASES_LIBRARY)
    target_link_libraries(mpc_core_v3 PUBLIC ${qpOASES_LIBRARY})
else()
    message(FATAL_ERROR "No qpOASES library target/path")
endif()

option(BUILD_PYTHON_BINDINGS "Build v3 Python modules" ON)
if(BUILD_PYTHON_BINDINGS)
    find_package(Python3 REQUIRED COMPONENTS Interpreter Development.Module)
    execute_process(COMMAND ${Python3_EXECUTABLE} -m pybind11 --cmakedir
        OUTPUT_VARIABLE PYBIND11_CMAKE_DIR OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
    if(PYBIND11_CMAKE_DIR)
        list(APPEND CMAKE_PREFIX_PATH "${PYBIND11_CMAKE_DIR}")
    endif()
    set(PYBIND11_FINDPYTHON ON)
    find_package(pybind11 CONFIG REQUIRED)
    pybind11_add_module(rokae_mpc_v3 python_bindings.cpp)
    target_link_libraries(rokae_mpc_v3 PRIVATE mpc_core_v3)
    set_target_properties(rokae_mpc_v3 PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/python")
endif()
include(CTest)
if(BUILD_TESTING)
    add_executable(test_preparation_feedback test_v3.cpp)
    target_link_libraries(test_preparation_feedback PRIVATE mpc_core_v3)
    add_test(NAME preparation_feedback COMMAND test_preparation_feedback)
endif()
