cmake_minimum_required(VERSION 3.18)
project(Robot_MPC_fpga LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
list(APPEND CMAKE_PREFIX_PATH "/opt/openrobots")
# 在父目录定位依赖，使导入目标对下面的CPU子目录和FPGA目标都可见。
find_package(Eigen3 REQUIRED)
find_package(pinocchio REQUIRED)
find_package(qpOASES REQUIRED)
find_package(Threads REQUIRED)
include(CTest)
# 共用同一套控制数学实现，只替换后台动力学后端。
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../src_v3" "${CMAKE_CURRENT_BINARY_DIR}/cpu_v3")

add_library(mpc_fpga_backend STATIC aba_codec.cpp xdma_device.cpp fpga_backend.cpp)
set_target_properties(mpc_fpga_backend PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_definitions(mpc_fpga_backend PRIVATE _FILE_OFFSET_BITS=64)
target_include_directories(mpc_fpga_backend PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/../include_fpga")
target_link_libraries(mpc_fpga_backend PUBLIC mpc_core_v3)
if(BUILD_PYTHON_BINDINGS)
    # pybind11的函数/目标由子目录加载；配置在当前目录再次定位，兼容非GLOBAL导入目标。
    find_package(Python3 REQUIRED COMPONENTS Interpreter Development.Module)
    execute_process(COMMAND ${Python3_EXECUTABLE} -m pybind11 --cmakedir
        OUTPUT_VARIABLE PYBIND11_CMAKE_DIR OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
    if(PYBIND11_CMAKE_DIR)
        list(APPEND CMAKE_PREFIX_PATH "${PYBIND11_CMAKE_DIR}")
    endif()
    set(PYBIND11_FINDPYTHON ON)
    find_package(pybind11 CONFIG REQUIRED)
    pybind11_add_module(rokae_mpc_fpga python_bindings.cpp)
    target_link_libraries(rokae_mpc_fpga PRIVATE mpc_fpga_backend)
    set_target_properties(rokae_mpc_fpga PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/python")
endif()
if(BUILD_TESTING)
    add_executable(test_fpga_mock test_fpga.cpp)
    target_link_libraries(test_fpga_mock PRIVATE mpc_fpga_backend)
    set(MPC_TEST_URDF "${CMAKE_CURRENT_SOURCE_DIR}/../urdf/ROKAE_SR4.urdf" CACHE FILEPATH "Test URDF")
    add_test(NAME fpga_codec_and_mock COMMAND test_fpga_mock "${MPC_TEST_URDF}")
endif()
